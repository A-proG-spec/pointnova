## app/withdraw/page.tsx

```tsx
'use client';

import { useEffect, useState } from 'react';
import { useRouter } from 'next/navigation';
import { Wallet, ArrowLeft, AlertCircle } from 'lucide-react';
import { useAuth } from '@/components/auth/AuthProvider';
import { LoadingSpinner } from '@/components/ui/LoadingSpinner';

export default function WithdrawPage() {
  const router = useRouter();
  const { user } = useAuth();
  const [userData, setUserData] = useState<any>(null);
  const [loading, setLoading] = useState(true);
  const [withdrawing, setWithdrawing] = useState(false);
  const [toast, setToast] = useState<{ message: string; type: 'success' | 'error' | 'info' } | null>(null);

  useEffect(() => {
    if (user) {
      fetchUserData();
    }
  }, [user]);

  const fetchUserData = async () => {
    try {
      const response = await fetch(`/api/user?userId=${user?.id}`);
      if (response.ok) {
        const data = await response.json();
        setUserData(data);
      }
    } catch (error) {
      console.error('Error fetching user data:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleWithdraw = async () => {
    if (!userData || userData.balance < 50) {
      setToast({
        message: 'Minimum withdrawal amount is 50 ETB',
        type: 'error'
      });
      return;
    }

    setWithdrawing(true);
    try {
      const response = await fetch('/api/withdraw', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          userId: user?.id,
          amount: userData.balance,
        }),
      });

      if (response.ok) {
        const data = await response.json();
        setToast({
          message: data.message || 'Withdrawal feature coming soon!',
          type: 'info'
        });
      }
    } catch (error) {
      console.error('Withdrawal error:', error);
      setToast({
        message: 'Failed to process withdrawal',
        type: 'error'
      });
    } finally {
      setWithdrawing(false);
    }
  };

  if (loading) {
    return <LoadingSpinner text="Loading..." />;
  }

  return (
    <div className="min-h-screen bg-black p-4 pb-24">
      {toast && (
        <div className={`fixed top-4 left-4 right-4 p-4 rounded-2xl z-50 animate-slide-down ${
          toast.type === 'error' ? 'bg-red-500/90' :
          toast.type === 'success' ? 'bg-green-500/90' :
          'bg-blue-500/90'
        }`}>
          <p className="text-white text-center font-semibold">{toast.message}</p>
        </div>
      )}

      <div className="flex items-center gap-4 mb-8">
        <button
          onClick={() => router.back()}
          className="p-2 hover:bg-white/10 rounded-full transition-all"
        >
          <ArrowLeft className="w-6 h-6 text-white" />
        </button>
        <h1 className="text-2xl font-bold text-white">Withdraw</h1>
        <Wallet className="w-6 h-6 text-purple-500 ml-auto" />
      </div>

      <div className="bg-gradient-to-br from-purple-500/20 to-purple-500/5 rounded-3xl p-8 border border-purple-500/20 mb-8">
        <p className="text-white/60 text-sm mb-2">Available Balance</p>
        <p className="text-4xl font-bold text-white mb-2">
          {userData?.balance || 0} ETB
        </p>
        <p className="text-white/40 text-sm">
          Total Earned: {userData?.totalEarned || 0} ETB
        </p>
      </div>

      <div className="bg-[#1a1a1a] rounded-2xl p-4 mb-6 border border-white/5">
        <div className="flex items-start gap-3">
          <AlertCircle className="w-5 h-5 text-yellow-500 flex-shrink-0 mt-0.5" />
          <div>
            <p className="text-white/80 text-sm font-semibold">Withdrawal Information</p>
            <ul className="text-white/60 text-sm mt-2 space-y-1">
              <li>• Minimum withdrawal: 50 ETB</li>
              <li>• Processing time: 24-48 hours</li>
              <li>• Withdrawal method: Coming soon</li>
            </ul>
          </div>
        </div>
      </div>

      <button
        onClick={handleWithdraw}
        disabled={withdrawing || !userData || userData.balance < 50}
        className={`w-full py-4 rounded-2xl font-semibold transition-all ${
          withdrawing || !userData || userData.balance < 50
            ? 'bg-white/5 text-white/40 cursor-not-allowed'
            : 'bg-gradient-to-r from-purple-500 to-purple-600 hover:from-purple-400 hover:to-purple-500 text-white shadow-lg shadow-purple-500/25 hover:scale-[1.02]'
        }`}
      >
        {withdrawing ? (
          <div className="flex items-center justify-center gap-2">
            <div className="animate-spin rounded-full h-5 w-5 border-t-2 border-b-2 border-white"></div>
            <span>Processing...</span>
          </div>
        ) : (
          'Withdraw Funds'
        )}
      </button>

      <div className="mt-6 grid grid-cols-2 gap-3">
        <div className="bg-[#1a1a1a] rounded-2xl p-4 border border-white/5">
          <p className="text-white/40 text-xs text-center">🔜 Bank Transfer</p>
        </div>
        <div className="bg-[#1a1a1a] rounded-2xl p-4 border border-white/5">
          <p className="text-white/40 text-xs text-center">🔜 Telebirr</p>
        </div>
      </div>
    </div>
  );
}
```

## app/login/page.tsx

```tsx
'use client';

import { useEffect, useState, useRef } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';
import { Coins, AlertCircle, Loader2 } from 'lucide-react';
import { useAuth } from '@/components/auth/AuthProvider';
import { useTelegram } from '@/components/telegram/TelegramProvider';

export default function LoginPage() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const { isAuthenticated, login } = useAuth();
  const { user: telegramUser, initData, isReady } = useTelegram();
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [debugLogs, setDebugLogs] = useState<string[]>([]);
  const authAttempted = useRef(false);

  const ref = searchParams.get('ref');

  // Utility to add debug logs
  const addLog = (message: string) => {
    console.log(message);
    setDebugLogs(prev => [...prev, `${new Date().toLocaleTimeString()}: ${message}`]);
  };

  useEffect(() => {
    if (isAuthenticated) {
      addLog('✅ User already authenticated, redirecting...');
      router.push('/');
    }
  }, [isAuthenticated, router]);

  useEffect(() => {
    // Auto-login with Telegram when ready
    if (isReady && initData && !isAuthenticated && !authAttempted.current) {
      addLog('🚀 Auto-login triggered: isReady=true, initData exists');
      authAttempted.current = true;
      handleTelegramLogin();
    }
  }, [isReady, initData, isAuthenticated]);

  const handleTelegramLogin = async () => {
    try {
      setIsLoading(true);
      setError(null);

      if (!initData) {
        const msg = 'No Telegram data available. Please open the app from Telegram.';
        addLog('❌ ' + msg);
        setError(msg);
        setIsLoading(false);
        return;
      }

      addLog(`📤 Sending login request with initData length: ${initData.length}`);
      addLog(`📎 Referral code: ${ref || 'none'}`);

      // Send to backend
      const response = await fetch('/api/auth/telegram', {
        method: 'POST',
          credentials: 'include',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ 
          initData,
          ref: ref || undefined,
        }),
      });

      addLog(`📨 Response status: ${response.status}`);

      // Get the actual response text
      const responseText = await response.text();
      addLog(`📝 Response length: ${responseText.length} characters`);
      addLog(`📋 Response preview: ${responseText.substring(0, 200)}`);

      // Check if response is HTML (error)
      if (responseText.startsWith('<!DOCTYPE') || responseText.startsWith('<html')) {
        addLog('❌ ERROR: Received HTML instead of JSON - API route may not exist or server error');
        throw new Error('Server returned HTML. Check that /api/auth/telegram route exists and is properly configured.');
      }

      // Try to parse as JSON
      let data;
      try {
        data = JSON.parse(responseText);
        addLog('✅ Successfully parsed JSON response');
      } catch (e) {
        addLog(`❌ Failed to parse JSON: ${e}`);
        throw new Error(`Invalid JSON response: ${responseText.substring(0, 100)}`);
      }

      if (!response.ok) {
        const errorMsg = data.error || 'Authentication failed';
        addLog(`❌ Auth failed: ${errorMsg}`);
        throw new Error(errorMsg);
      }

      addLog(`✅ Auth successful, user ID: ${data.user?.id}`);
      
      // Login user
      login(data.user);
      
      addLog('🔄 Redirecting to home page...');
      router.push('/');
    } catch (error) {
      const errorMsg = error instanceof Error ? error.message : 'Failed to login';
      addLog(`❌ Login error: ${errorMsg}`);
      console.error('Login error:', error);
      setError(errorMsg);
      setIsLoading(false);
    }
  };

  // Check if in Telegram
  const isInTelegram = typeof window !== 'undefined' && !!(window as any).Telegram?.WebApp;

  // If not in Telegram, show alternative login
  if (!isInTelegram && process.env.NODE_ENV !== 'development') {
    return (
      <div className="min-h-screen bg-black flex items-center justify-center p-4">
        <div className="w-full max-w-md text-center">
          <div className="inline-block p-4 bg-green-500/10 rounded-full mb-4">
            <Coins className="w-12 h-12 text-green-500" />
          </div>
          <h1 className="text-3xl font-bold text-white mb-2">PointNova</h1>
          <p className="text-white/60 mb-8">Earn rewards by completing tasks</p>
          
          <div className="bg-[#1a1a1a] rounded-2xl p-8 border border-white/5">
            <h2 className="text-xl font-semibold text-white mb-4">Open in Telegram</h2>
            <p className="text-white/60 text-sm mb-6">
              This app is designed to work within Telegram. 
              Please open it from the Telegram app.
            </p>
            <button
              onClick={() => {
                const botUsername = process.env.NEXT_PUBLIC_TELEGRAM_BOT_USERNAME || 'your_bot_username';
                window.location.href = `tg://resolve?domain=${botUsername}&startapp=referral`;
              }}
              className="w-full bg-blue-500 hover:bg-blue-400 text-white font-semibold py-3 rounded-xl transition-all"
            >
              Open in Telegram
            </button>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-black flex items-center justify-center p-4">
      <div className="w-full max-w-md">
        {/* Logo */}
        <div className="text-center mb-8">
          <div className="inline-block p-4 bg-green-500/10 rounded-full mb-4">
            <Coins className="w-12 h-12 text-green-500" />
          </div>
          <h1 className="text-3xl font-bold text-white">PointNova</h1>
          <p className="text-white/60 mt-2">Earn rewards by completing tasks</p>
        </div>

        {/* Login Card */}
        <div className="bg-[#1a1a1a] rounded-2xl p-8 border border-white/5">
          <h2 className="text-xl font-semibold text-white mb-6">
            {isLoading ? 'Authenticating...' : 'Welcome Back'}
          </h2>

          {error && (
            <div className="mb-4 p-3 bg-red-500/10 border border-red-500/20 rounded-xl flex items-start gap-2">
              <AlertCircle className="w-4 h-4 text-red-400 flex-shrink-0 mt-0.5" />
              <div className="flex-1">
                <p className="text-red-400 text-sm font-semibold mb-1">Error</p>
                <p className="text-red-400 text-xs">{error}</p>
              </div>
            </div>
          )}

          {!isReady && (
            <div className="mb-4 p-3 bg-yellow-500/10 border border-yellow-500/20 rounded-xl flex items-start gap-2">
              <Loader2 className="w-4 h-4 text-yellow-400 flex-shrink-0 mt-0.5 animate-spin" />
              <p className="text-yellow-400 text-sm">Loading Telegram data...</p>
            </div>
          )}

          {telegramUser && (
            <div className="mb-4 p-3 bg-green-500/10 border border-green-500/20 rounded-xl flex items-center gap-3">
              {telegramUser.photo_url ? (
                <img 
                  src={telegramUser.photo_url} 
                  alt={telegramUser.first_name}
                  className="w-8 h-8 rounded-full"
                />
              ) : (
                <div className="w-8 h-8 rounded-full bg-blue-500/20 flex items-center justify-center text-blue-500 font-bold">
                  {telegramUser.first_name?.charAt(0) || '?'}
                </div>
              )}
              <div>
                <p className="text-white font-semibold text-sm">
                  {telegramUser.first_name} {telegramUser.last_name || ''}
                </p>
                {telegramUser.username && (
                  <p className="text-white/40 text-xs">@{telegramUser.username}</p>
                )}
              </div>
            </div>
          )}

          <button
            onClick={handleTelegramLogin}
            disabled={isLoading || !initData || !isReady}

                </svg>
                <span>Continue with Telegram</span>
              </>
            )}
          </button>

          {!isReady && (
            <p className="mt-3 text-center text-yellow-500 text-xs">
              ⏳ Waiting for Telegram initialization...
            </p>
          )}

          {isReady && !initData && (
            <p className="mt-3 text-center text-yellow-500 text-xs">
              ⚠️ No Telegram data detected. Make sure you're opening this from Telegram.
            </p>
          )}

          {process.env.NODE_ENV === 'development' && (
            <div className="mt-4 pt-4 border-t border-white/5">
              <button
                onClick={() => window.location.href = '/api/auth/demo'}
                className="w-full bg-yellow-500/10 hover:bg-yellow-500/20 text-yellow-500 font-semibold py-2 rounded-xl transition-all text-sm"

                🚀 Quick Demo Login (Dev Only)

          <div className="mt-4 text-center">
            <p className="text-white/40 text-xs">
              By continuing, you agree to our Terms of Service
            </p>
          </div>
        </div>

        {/* Debug Logs - Only in development */}
        {process.env.NODE_ENV === 'development' && debugLogs.length > 0 && (
```

```tsx
'use client';

import { useEffect, useState } from 'react';
import { useRouter } from 'next/navigation';
import { Copy, Share2, Users, ArrowLeft, Check } from 'lucide-react';
import { useAuth } from '@/components/auth/AuthProvider';
import { useTelegram } from '@/components/telegram/TelegramProvider';
import { LoadingSpinner } from '@/components/ui/LoadingSpinner';

interface ReferralData {
  referralCode: string;
  referralLink: string;
  referralCount: number;
}

export default function InvitePage() {
  const router = useRouter();
  const { user } = useAuth();
  const { webApp } = useTelegram();
  const [referralData, setReferralData] = useState<ReferralData | null>(null);
  const [loading, setLoading] = useState(true);
  const [copied, setCopied] = useState(false);

  useEffect(() => {
    if (user) {
      fetchReferralData();
    }
  }, [user]);

  const fetchReferralData = async () => {
    try {
      const response = await fetch(`/api/referral?userId=${user?.id}`);
      if (response.ok) {
        const data = await response.json();
        setReferralData(data);
      }
    } catch (error) {
      console.error('Error fetching referral data:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleCopy = async () => {
    if (referralData?.referralLink) {
      try {
        await navigator.clipboard.writeText(referralData.referralLink);
        setCopied(true);
        setTimeout(() => setCopied(false), 2000);
      } catch (error) {
        console.error('Error copying:', error);
      }
    }
  };

  const handleShare = async () => {
    // Try Telegram share first
    if (webApp) {
      try {
        webApp.sendData(JSON.stringify({
          type: 'share',
          text: `Join PointNova and start earning rewards! Use my referral link: ${referralData?.referralLink}`,
        }));
        return;
      } catch (error) {
        console.error('Telegram share error:', error);
      }
    }

    // Fallback to Web Share API
    if (referralData?.referralLink) {
      try {
        await navigator.share({
          title: 'Join PointNova',
          text: `Join PointNova and start earning rewards! Use my referral link: ${referralData.referralLink}`,
          url: referralData.referralLink,
        });
      } catch (error) {
        console.error('Error sharing:', error);
        handleCopy();
      }
    }
  };

  if (loading) {
    return <LoadingSpinner text="Loading..." />;
  }

  return (
    <div className="min-h-screen bg-black p-4 pb-24">
      <div className="flex items-center gap-4 mb-8">
        <button
          onClick={() => router.back()}
          className="p-2 hover:bg-white/10 rounded-full transition-all"
        >
          <ArrowLeft className="w-6 h-6 text-white" />
        </button>
        <h1 className="text-2xl font-bold text-white">Invite Friends</h1>
        <Users className="w-6 h-6 text-blue-500 ml-auto" />
      </div>

      <div className="bg-gradient-to-br from-[#1a1a1a] to-[#0d0d0d] rounded-3xl p-8 border border-white/5">
        <div className="text-center mb-8">
          <div className="w-20 h-20 bg-blue-500/10 rounded-full flex items-center justify-center mx-auto mb-4">
            <Users className="w-10 h-10 text-blue-500" />
          </div>
          <h2 className="text-2xl font-bold text-white mb-2">Invite & Earn</h2>
          <p className="text-white/60 text-sm">
            Share your referral link with friends and earn rewards when they join!
          </p>
        </div>

        <div className="bg-black/50 rounded-2xl p-4 mb-6">
          <div className="flex items-center justify-between">
            <span className="text-white/60 text-sm">Total Referrals</span>
            <span className="text-white font-bold text-xl">{referralData?.referralCount || 0}</span>
          </div>
        </div>

        <div className="bg-black/50 rounded-2xl p-4 mb-4">
          <p className="text-white/60 text-sm mb-2">Your Referral Code</p>
          <p className="text-white font-mono text-lg font-bold tracking-wider">
            {referralData?.referralCode}
          </p>
        </div>

        <div className="bg-black/50 rounded-2xl p-4 mb-6">
          <p className="text-white/60 text-sm mb-2">Your Referral Link</p>
          <div className="flex items-center gap-2">
            <p className="text-white/80 text-sm flex-1 break-all font-mono">
              {referralData?.referralLink}
            </p>
          </div>
        </div>

        <div className="space-y-3">
          <button
            onClick={handleCopy}
            className="w-full bg-white/5 hover:bg-white/10 text-white font-semibold py-4 rounded-2xl flex items-center justify-center gap-2 transition-all border border-white/10"
          >
            {copied ? (
              <>
                <Check className="w-5 h-5 text-green-500" />
                <span>Copied!</span>
              </>
            ) : (
              <>
                <Copy className="w-5 h-5" />
                <span>Copy Link</span>
              </>
            )}
          </button>
          
          <button
            onClick={handleShare}
            className="w-full bg-blue-500 hover:bg-blue-400 text-white font-semibold py-4 rounded-2xl flex items-center justify-center gap-2 transition-all"
          >
            <Share2 className="w-5 h-5" />
            <span>Share Link</span>
          </button>
        </div>
      </div>

      <div className="mt-6 bg-[#1a1a1a] rounded-2xl p-4 border border-white/5">
        <h3 className="text-white font-semibold mb-2">How it works</h3>
        <div className="space-y-2 text-white/60 text-sm">
          <p>1. Share your referral link with friends</p>
          <p>2. Friends join PointNova using your link</p>

  );
```
